## 🔧 数据操作接口

### 📖 **数据读取接口**
```javascript
// 获取所有用户
const users = require('@/data/users.js');

// 获取当前登录用户
const currentUser = users.find(user => user.isLoggedIn === true);

// 获取用户特定数据
const userBalance = currentUser.balance;
const transferRecords = currentUser.transferRecords;
```

### ✏️ **数据更新接口**
```javascript
// 更新用户数据
const updateUserData = (userId, newData) => {
  const userIndex = users.findIndex(user => user.id === userId);
  if (userIndex !== -1) {
    users[userIndex] = { ...users[userIndex], ...newData };
    return true;
  }
  return false;
};

// 添加新记录
const addTransferRecord = (userId, record) => {
  const user = users.find(user => user.id === userId);
  if (user) {
    user.transferRecords.push(record);
    return true;
  }
  return false;
};
```
