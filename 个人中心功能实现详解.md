## 1. 个人资料管理

### 功能描述
个人资料管理是用户中心的核心功能，包括用户基本信息编辑、数据同步、表单验证等。

### 实现方式

#### 1.1 数据结构设计
```javascript
// 用户资料数据结构
const profileData = {
  avatar: '',           // 头像URL
  realName: '',         // 真实姓名
  nickname: '',         // 昵称
  gender: '男',         // 性别
  birthDate: '',        // 出生日期
  phone: '',           // 手机号码
  email: '',           // 邮箱地址
  idCard: '',          // 身份证号
  address: ''          // 居住地址
}
```

#### 1.2 表单验证机制
```javascript
// 表单验证实现
validateForm() {
  const errors = []
  
  // 真实姓名验证
  if (!this.profileData.realName || this.profileData.realName.length < 2) {
    errors.push('真实姓名至少2个字符')
  }
  
  // 手机号验证
  if (!/^1[3-9]\d{9}$/.test(this.profileData.phone)) {
    errors.push('请输入正确的手机号码')
  }
  
  // 邮箱验证
  if (this.profileData.email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(this.profileData.email)) {
    errors.push('请输入正确的邮箱地址')
  }
  
  return errors
}
```

#### 1.3 数据保存和同步
```javascript
// 保存个人资料
async saveProfile() {
  try {
    uni.showLoading({ title: '保存中...' })
    
    // 表单验证
    const errors = this.validateForm()
    if (errors.length > 0) {
      uni.showToast({ title: errors[0], icon: 'none' })
      return
    }
    
    // 获取当前用户信息
    let userInfo = uni.getStorageSync('userInfo') || uni.getStorageSync('currentUser')
    if (userInfo) {
      // 更新用户信息
      userInfo = { 
        ...userInfo, 
        ...this.profileData,
        lastUpdateTime: new Date().toISOString()
      }
      
      // 保存到本地存储
      uni.setStorageSync('userInfo', userInfo)
      uni.setStorageSync('currentUser', userInfo)
      
      // 更新本地数据库
      await this.updateLocalDatabase(userInfo)
      
      // 同步到服务器
      try {
        await this.syncProfileToServer(userInfo)
        console.log('个人资料同步成功')
      } catch (syncError) {
        console.error('同步失败，已加入同步队列:', syncError)
        syncManager.addSyncTask('updateProfile', userInfo)
      }
    }
    
    uni.hideLoading()
    uni.showToast({ title: '保存成功', icon: 'success' })
    
  } catch (error) {
    console.error('保存个人资料失败:', error)
    uni.hideLoading()
    uni.showToast({ title: '保存失败', icon: 'none' })
  }
}
```

### 遇到的问题及解决方案

#### 问题1：数据同步不一致
**问题描述：** 用户修改资料后，重新登录时数据丢失

**解决方案：**
```javascript
// 数据一致性检查
export function checkAndFixUserDataConsistency() {
  const currentUserId = uni.getStorageSync('currentUserId')
  const users = uni.getStorageSync('users') || []
  const userInfo = uni.getStorageSync('userInfo')
  
  const dbUser = users.find(user => user.id === currentUserId)
  
  // 检查数据一致性
  const isConsistent = userInfo && 
    userInfo.id === dbUser.id &&
    userInfo.username === dbUser.username &&
    userInfo.phone === dbUser.phone
  
  if (!isConsistent) {
    // 数据不一致，进行修复
    const fixedUserInfo = {
      ...userInfo, // 保持当前用户信息
      ...dbUser,   // 用数据库数据覆盖
      isLoggedIn: true
    }
    
    uni.setStorageSync('userInfo', fixedUserInfo)
    uni.setStorageSync('currentUser', fixedUserInfo)
  }
}
```

#### 问题2：表单验证复杂
**问题描述：** 不同字段的验证规则复杂，用户体验差

```javascript
// 处理选中的图片
async handleImageSelected(imagePath) {
  uni.showLoading({ title: '上传中...' })
  
  try {
    // 检查网络状态
    const networkRes = await this.checkNetworkStatus()
    
    if (networkRes.networkType === 'none') {
      // 离线模式，先保存到本地
      this.profileData.avatar = imagePath
      await this.saveAvatarToStorage(imagePath)
      uni.hideLoading()
      uni.showToast({ title: '头像已保存，网络恢复后将自动上传', icon: 'success' })
      return
    }
    
    // 在线模式，上传到服务器
    const uploadResult = await this.uploadAvatarToServer(imagePath)
    
    if (uploadResult.success) {
      this.profileData.avatar = uploadResult.avatarUrl
      await this.saveAvatarToStorage(uploadResult.avatarUrl)
      uni.hideLoading()
      uni.showToast({ title: '头像更新成功', icon: 'success' })
    } else {
      throw new Error(uploadResult.message || '上传失败')
    }
  } catch (error) {
    console.error('头像上传失败:', error)
    uni.hideLoading()
    uni.showToast({ title: error.message || '头像上传失败', icon: 'none' })
  }
}
```

#### 2.4 服务器上传实现
```javascript
// 上传头像到服务器
uploadAvatarToServer(imagePath) {
  return new Promise((resolve, reject) => {
    uni.uploadFile({
      url: 'https://api.abchina.com/user/avatar',
      filePath: imagePath,
      name: 'avatar',
      header: {
        'Authorization': `Bearer ${uni.getStorageSync('token') || ''}`
      },
      success: (res) => {
        try {
          const data = JSON.parse(res.data)
          if (data.code === 0) {
            resolve({
              success: true,
              avatarUrl: data.data.avatarUrl
            })
          } else {
            reject(new Error(data.message || '上传失败'))
          }
        } catch (parseError) {
          reject(new Error('服务器响应格式错误'))
        }
      },
      fail: (error) => {
        console.error('上传失败:', error)
        // 上传失败时，先保存到本地
        resolve({
          success: true,
          avatarUrl: imagePath,
          isOffline: true,
          message: '网络异常，已保存到本地'
        })
      }
    })
  })
}
```

compressImage(imagePath) {

      src: imagePath,
      quality: 80, // 压缩质量

        resolve(res.tempFilePath)

      fail: () => {

      }

// 应用主题
applyTheme(theme) {
  try {
    this.currentTheme = theme
    this.setStoredTheme(theme)
    
    const actualTheme = theme === 'system' ? this.getSystemTheme() : theme
    const config = this.getThemeConfig(actualTheme)
    
    // 设置CSS变量
    this.setCSSVariables(config.colors)
    
    // 设置页面类名
    this.setPageClasses(actualTheme)
    
    // 触发主题变化事件
    this.triggerThemeChange(theme, config)
    
    console.log('主题已应用:', theme, '实际主题:', actualTheme)
  } catch (error) {
    console.error('应用主题失败:', error)
  }
}

// 设置CSS变量
setCSSVariables(colors) {
  try {
    if (typeof document !== 'undefined' && document.documentElement) {
      const root = document.documentElement
      
      Object.entries(colors).forEach(([key, value]) => {
        root.style.setProperty(`--theme-${key}`, value)
      })
      
      // 设置组合变量
      root.style.setProperty('--theme-glass-bg', colors.cardBackground + '80')
      root.style.setProperty('--theme-glass-border', colors.borderColor + '40')
      root.style.setProperty('--theme-shadow-sm', `0 2rpx 8rpx ${colors.shadowColor}`)
      root.style.setProperty('--theme-shadow-md', `0 4rpx 12rpx ${colors.shadowColor}`)
      root.style.setProperty('--theme-shadow-lg', `0 8rpx 24rpx ${colors.shadowColor}`)
    }
  } catch (error) {
    console.error('设置CSS变量失败:', error)
  }
}

// 获取系统主题
getSystemTheme() {
  try {
    if (typeof window !== 'undefined' && window.matchMedia) {
      return window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light'
    }
  } catch (error) {
    console.error('获取系统主题失败:', error)
  }
  return 'light'
}

// 设置系统主题监听器
setupSystemThemeListener() {
  try {
    if (typeof window !== 'undefined' && window.matchMedia) {
      const mediaQuery = window.matchMedia('(prefers-color-scheme: dark)')
      
      const handleSystemThemeChange = (e) => {
        if (this.currentTheme === 'system') {
          this.applyTheme('system')
        }
      }
      
      mediaQuery.addEventListener('change', handleSystemThemeChange)
    }
  } catch (error) {
    console.error('设置系统主题监听器失败:', error)
  }
}
```

// 处理拨打电话错误
handlePhoneCallError(error) {
  let errorMessage = '拨打电话失败'
  
  switch (error.errMsg) {
    case 'makePhoneCall:fail cancel':
      errorMessage = '用户取消拨打电话'
      break
    case 'makePhoneCall:fail no permission':
      errorMessage = '没有拨打电话的权限，请在设置中开启'
      break
    case 'makePhoneCall:fail system error':
      errorMessage = '系统错误，请稍后重试'
      break
    default:
      if (error.errMsg && error.errMsg.includes('permission')) {
        errorMessage = '没有拨打电话的权限'
      }
  }
  
  uni.showModal({
    title: '拨打电话失败',
    content: errorMessage,
    showCancel: false,
    confirmText: '确定'
  })
}
```

// 统一权限检查接口
async checkPhonePermission() {
  try {
    const systemInfo = uni.getSystemInfoSync()
    
    // 根据平台进行不同的权限检查
    if (systemInfo.platform === 'android') {
      return await this.checkAndroidPermission()
    } else if (systemInfo.platform === 'ios') {
      return await this.checkIOSPermission()
    } else {
      // 其他平台默认允许
      return true
    }
  } catch (error) {
    console.error('权限检查失败:', error)
    return false
  }
}

// Android权限检查
checkAndroidPermission() {
  return new Promise((resolve) => {
    if (typeof plus !== 'undefined') {
      plus.android.requestPermissions(
        ['android.permission.CALL_PHONE'],
        (result) => {
          if (result.granted && result.granted.length > 0) {
            resolve(true)
          } else {
            // 权限被拒绝，引导用户到设置页面
            this.showPermissionGuide()
            resolve(false)
          }
        },
        (error) => {
          console.error('Android权限检查失败:', error)
          resolve(false)
        }
      )
    } else {
      resolve(true)
    }
  })
}
